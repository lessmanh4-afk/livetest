ScreenCast Live - Android Studio project
=========================================
Open this folder in Android Studio (Koala or newer, JDK 17). Let Gradle sync.

BEFORE BUILDING (required)
1. Firebase console -> create project -> add Android app with package name:  com.screencastlive
2. Download google-services.json and put it in:  app/google-services.json
3. Build -> Authentication -> Sign-in method -> enable "Anonymous"
4. Build -> Realtime Database -> Create database -> Rules tab -> paste database.rules.json -> Publish
5. Run on two phones (or install the APK). Phone A: Broadcast tab -> Start. Phone B (mobile data): Public tab -> tap the stream.

TURN: Metered credentials are in Core.kt (Config). Regenerate them in the Metered dashboard before public release.
If anything fails, open Broadcast tab -> "Copy log" and send the log.

How the "Failed to start MediaProjection" bug is avoided
- Permission result Intent -> ScreenBroadcastService -> startForeground(type=mediaProjection) FIRST -> only then ScreenCapturerAndroid.startCapture.
- Manifest service has foregroundServiceType="mediaProjection".
- Service is START_NOT_STICKY (Android 14 projection Intent is single use). Reconnects use ICE restart, never a new permission.

package com.screencastlive

import android.content.Context
import android.content.Intent
import android.media.projection.MediaProjection
import android.net.ConnectivityManager
import android.net.Network
import android.os.Handler
import android.os.Looper

import com.google.firebase.database.DataSnapshot
import com.google.firebase.database.DatabaseError
import com.google.firebase.database.FirebaseDatabase
import com.google.firebase.database.ServerValue
import com.google.firebase.database.ValueEventListener

import org.webrtc.MediaConstraints
import org.webrtc.PeerConnection
import org.webrtc.ScreenCapturerAndroid
import org.webrtc.SessionDescription
import org.webrtc.SurfaceTextureHelper
import org.webrtc.VideoSource
import org.webrtc.VideoTrack
import java.util.concurrent.ConcurrentHashMap

/**
 * Captures the screen and streams it to viewers over WebRTC.
 * Stream id == owner uid. Signaling lives under signaling/{uid}/viewers/{viewerUid}.
 * Must be created and started from ScreenBroadcastService AFTER startForeground(mediaProjection).
 */
class Broadcaster(
    private val ctx: Context,
    private val uid: String,
    private val title: String,
    private val projectionData: Intent,
    private val onStopped: () -> Unit
) {
    private val db = FirebaseDatabase.getInstance()
    private val streamRef = db.getReference("streams/$uid")
    private val sigRoot = db.getReference("signaling/$uid")
    private val sigRef = db.getReference("signaling/$uid/viewers")
    private val handler = Handler(Looper.getMainLooper())

    private var capturer: ScreenCapturerAndroid? = null
    private var helper: SurfaceTextureHelper? = null
    private var source: VideoSource? = null
    private var track: VideoTrack? = null
    private val peers = ConcurrentHashMap<String, Peer>()
    private var childL: ChildEventListener? = null
    private var connL: ValueEventListener? = null
    private var netCb: ConnectivityManager.NetworkCallback? = null
    @Volatile private var netLost = false
    @Volatile private var stopped = false

    fun start() {
        Rtc.init(ctx)

        AppLog.log("Step 1: creating ScreenCapturerAndroid")
        val cap = ScreenCapturerAndroid(projectionData, object : MediaProjection.Callback() {
            override fun onStop() {
                AppLog.log("MediaProjection stopped (user or system)")
                handler.post { if (!stopped) onStopped() }
            }
        })
        capturer = cap
        helper = SurfaceTextureHelper.create("ScreenCaptureThread", Rtc.eglBase.eglBaseContext)
        val src = Rtc.factory.createVideoSource(true)
        source = src
        cap.initialize(helper, ctx, src.capturerObserver)

        val (w, h) = captureSize()
        AppLog.log("Step 2: startCapture ${w}x$h @24fps")
        cap.startCapture(w, h, 24)
        AppLog.log("Step 2 OK: capture started")

        track = Rtc.factory.createVideoTrack("screen", src).also { it.setEnabled(true) }

        AppLog.log("Step 3: publishing stream in Firebase")
        sigRoot.removeValue()
        streamRef.setValue(
            mapOf(
                "ownerUid" to uid, "title" to title, "status" to "live",
                "startedAt" to ServerValue.TIMESTAMP, "viewerCount" to 0
            )
        )

        // Keep status correct across reconnects: whenever Firebase (re)connects, mark live
        // and re-arm the server-side onDisconnect -> offline.
        connL = object : ValueEventListener {
            override fun onDataChange(s: DataSnapshot) {
                if (s.getValue(Boolean::class.java) == true && !stopped) {
                    AppLog.log("Firebase connected -> status live")
                    streamRef.child("status").onDisconnect().setValue("offline")
                    streamRef.child("status").setValue("live")
                } else if (!stopped) {
                    AppLog.log("Firebase disconnected")
                }
            }
            override fun onCancelled(e: DatabaseError) {}
        }
        db.getReference(".info/connected").addValueEventListener(connL!!)

        childL = object : ChildEventListener {
            override fun onChildAdded(s: DataSnapshot, prev: String?) {
                val k = s.key ?: return
                addPeer(k)
            }
            override fun onChildRemoved(s: DataSnapshot) {
                val k = s.key ?: return
                removePeer(k)
            }
            override fun onChildChanged(s: DataSnapshot, prev: String?) {}
            override fun onChildMoved(s: DataSnapshot, prev: String?) {}
            override fun onCancelled(e: DatabaseError) { AppLog.log("signaling cancelled: ${e.message}") }
        }
        sigRef.addChildEventListener(childL!!)

        registerNetwork()
        BroadcastState.state.value = "Live"
        AppLog.log("Broadcast is LIVE")
    }

    private fun captureSize(): Pair<Int, Int> {
        val dm = ctx.resources.displayMetrics
        val scale = minOf(1f, 1280f / maxOf(dm.widthPixels, dm.heightPixels))
        val w = ((dm.widthPixels * scale).toInt() / 2) * 2
        val h = ((dm.heightPixels * scale).toInt() / 2) * 2
        return Pair(w, h)
    }

    private fun registerNetwork() {
        val cm = ctx.getSystemService(ConnectivityManager::class.java)
        val cb = object : ConnectivityManager.NetworkCallback() {
            override fun onLost(network: Network) {
                netLost = true
                AppLog.log("Network lost")
                BroadcastState.state.value = "Reconnecting"
            }
            override fun onAvailable(network: Network) {
                if (!netLost) return
                netLost = false
                handler.post {
                    if (stopped) return@post
                    AppLog.log("Network back -> ICE restart for ${peers.size} viewer(s)")
                    peers.values.forEach { it.restart() }
                    BroadcastState.state.value = "Live"
                }
            }
        }
        netCb = cb
        cm.registerDefaultNetworkCallback(cb)
    }

    private fun addPeer(vid: String) {
        if (stopped) return
        peers.remove(vid)?.close()
        if (peers.size >= Config.MAX_VIEWERS) {
            AppLog.log("Viewer limit reached, rejecting $vid")
            sigRef.child(vid).child("error").setValue("full")
            return
        }
        val p = Peer(vid)
        peers[vid] = p
        p.start()
        updateCount()
    }

    private fun removePeer(vid: String) {
        peers.remove(vid)?.close()
        updateCount()
    }

    private fun updateCount() {
        BroadcastState.viewers.value = peers.size
        if (!stopped) streamRef.child("viewerCount").setValue(peers.size)
    }

    fun stop() {
        if (stopped) return
        stopped = true
        AppLog.log("Stopping broadcast")
        try { netCb?.let { ctx.getSystemService(ConnectivityManager::class.java).unregisterNetworkCallback(it) } } catch (_: Throwable) {}
        connL?.let { db.getReference(".info/connected").removeEventListener(it) }
        childL?.let { sigRef.removeEventListener(it) }
        peers.values.forEach { it.close() }
        peers.clear()
        try { capturer?.stopCapture() } catch (e: Throwable) { AppLog.log("stopCapture: ${e.message}") }
        try { capturer?.dispose() } catch (_: Throwable) {}
        try { track?.dispose() } catch (_: Throwable) {}
        try { source?.dispose() } catch (_: Throwable) {}
        try { helper?.dispose() } catch (_: Throwable) {}
        streamRef.child("status").setValue("offline")
        streamRef.child("status").onDisconnect().cancel()
        streamRef.child("viewerCount").setValue(0)
        sigRoot.removeValue()
        BroadcastState.viewers.value = 0
    }

    /** One PeerConnection per viewer. */
    private inner class Peer(val vid: String) {
        private val ref = sigRef.child(vid)
        private var pc: PeerConnection? = null
        private var n = 0
        private var remoteSet = false
        private val pending = ArrayList<IceCandidate>()
        private var answerL: ValueEventListener? = null
        private var candL: ChildEventListener? = null
        private var attempts = 0
        private val restartRunnable = Runnable { attempts++; restart() }

        fun start() {
            AppLog.log("[$vid] new viewer")
            val conn = Rtc.factory.createPeerConnection(Rtc.rtcConfig(), object : PcObserver() {
                override fun onIceCandidate(c: IceCandidate?) {
                    c ?: return
                    ref.child("bc").push().setValue(
                        mapOf("mid" to c.sdpMid, "idx" to c.sdpMLineIndex, "sdp" to c.sdp)
                    )
                }
                override fun onIceConnectionChange(s: PeerConnection.IceConnectionState?) {
                    AppLog.log("[$vid] ICE $s")
                    handler.post {
                        when (s) {
                            PeerConnection.IceConnectionState.CONNECTED,
                            PeerConnection.IceConnectionState.COMPLETED -> {
                                attempts = 0
                                handler.removeCallbacks(restartRunnable)
                            }
                            PeerConnection.IceConnectionState.DISCONNECTED -> scheduleRestart(5000)
                            PeerConnection.IceConnectionState.FAILED -> scheduleRestart(
                                minOf(2000L shl minOf(attempts, 4), 30000L)
                            )
                            else -> {}
                        }
                    }
                }
            })
            if (conn == null) { AppLog.log("[$vid] createPeerConnection returned null"); return }
            pc = conn

            val sender = conn.addTrack(track, listOf("screen"))
            try {
                val params = sender.parameters
                params.encodings.forEach { it.maxBitrateBps = 2_500_000; it.maxFramerate = 24 }
                sender.setParameters(params)
            } catch (e: Throwable) { AppLog.log("[$vid] bitrate params: ${e.message}") }

            answerL = object : ValueEventListener {
                override fun onDataChange(s: DataSnapshot) {
                    val sdp = s.child("sdp").getValue(String::class.java) ?: return
                    val an = (s.child("n").value as? Number)?.toInt() ?: return
                    val p = pc ?: return
                    if (an != n || p.signalingState() != PeerConnection.SignalingState.HAVE_LOCAL_OFFER) return
                    p.setRemoteDescription(object : SdpAdapter("setRemote(answer)") {
                        override fun onSetSuccess() {
                            AppLog.log("[$vid] answer applied")
                            synchronized(pending) {
                                remoteSet = true
                                pending.forEach { p.addIceCandidate(it) }
                                pending.clear()
                            }
                        }
                    }, SessionDescription(SessionDescription.Type.ANSWER, sdp))
                }
                override fun onCancelled(e: DatabaseError) {}
            }
            ref.child("answer").addValueEventListener(answerL!!)

            candL = object : ChildEventListener {
                override fun onChildAdded(s: DataSnapshot, prev: String?) {
                    val sdp = s.child("sdp").getValue(String::class.java) ?: return
                    val mid = s.child("mid").getValue(String::class.java)
                    val idx = (s.child("idx").value as? Number)?.toInt() ?: 0
                    val cand = IceCandidate(mid, idx, sdp)
                    synchronized(pending) {
                        if (remoteSet) pc?.addIceCandidate(cand) else pending.add(cand)
                    }
                }
                override fun onChildChanged(s: DataSnapshot, prev: String?) {}
                override fun onChildRemoved(s: DataSnapshot) {}
                override fun onChildMoved(s: DataSnapshot, prev: String?) {}
                override fun onCancelled(e: DatabaseError) {}
            }
            ref.child("vc").addChildEventListener(candL!!)

            makeOffer(false)
        }

        private fun scheduleRestart(delayMs: Long) {
            handler.removeCallbacks(restartRunnable)
            handler.postDelayed(restartRunnable, delayMs)
        }

        fun restart() {
            if (stopped) return
            AppLog.log("[$vid] ICE restart")
            makeOffer(true)
        }

        private fun makeOffer(iceRestart: Boolean) {
            val p = pc ?: return
            val myN = ++n
            val c = MediaConstraints()
            if (iceRestart) c.mandatory.add(MediaConstraints.KeyValuePair("IceRestart", "true"))
            p.createOffer(object : SdpAdapter("createOffer") {
                override fun onCreateSuccess(d: SessionDescription?) {
                    d ?: return
                    p.setLocalDescription(object : SdpAdapter("setLocal(offer)") {
                        override fun onSetSuccess() {
                            ref.child("offer").setValue(mapOf("sdp" to d.description, "n" to myN))
                            AppLog.log("[$vid] offer #$myN sent")
                        }
                    }, d)
                }
            }, c)
        }

        fun close() {
            handler.removeCallbacks(restartRunnable)
            answerL?.let { ref.child("answer").removeEventListener(it) }
            candL?.let { ref.child("vc").removeEventListener(it) }
            try { pc?.close(); pc?.dispose() } catch (_: Throwable) {}
            pc = null
            AppLog.log("[$vid] closed")
        }
    }
}

            if (r.resultCode == RESULT_OK && data != null) {
                AppLog.log("Screen capture permission granted -> starting service")
                val i = Intent(this, ScreenBroadcastService::class.java)
                    .setAction(ScreenBroadcastService.ACTION_START)
                    .putExtra(ScreenBroadcastService.EXTRA_DATA, data)
                    .putExtra(ScreenBroadcastService.EXTRA_TITLE, pendingTitle)
                startForegroundService(i)
            } else {
                AppLog.log("Screen capture permission denied")
            }
        }

    override fun onCreate(savedInstanceState: Bundle?) {
        super.onCreate(savedInstanceState)
        Rtc.init(applicationContext)
        signIn()
        setContent {
            MaterialTheme {
                Surface(Modifier.fillMaxSize()) {
                    AppUi(uid, authError, ::startBroadcast, ::stopBroadcast)
                }
            }
        }
    }

    private fun signIn() {
        val auth = FirebaseAuth.getInstance()
        val u = auth.currentUser
        if (u != null) { uid = u.uid; return }
        auth.signInAnonymously()
            .addOnSuccessListener { uid = it.user?.uid; AppLog.log("Signed in as guest") }
            .addOnFailureListener { authError = "Sign-in failed: ${it.message}"; AppLog.log("Auth failed: ${it.message}") }
    }

    private fun startBroadcast(title: String) {
        pendingTitle = title
        if (Build.VERSION.SDK_INT >= 33 &&
            checkSelfPermission(Manifest.permission.POST_NOTIFICATIONS) != PackageManager.PERMISSION_GRANTED
        ) {
            notifLauncher.launch(Manifest.permission.POST_NOTIFICATIONS)
        } else {
            askBattery()
        }
    }

    private fun askBattery() {
        val pm = getSystemService(POWER_SERVICE) as PowerManager
        if (pm.isIgnoringBatteryOptimizations(packageName)) { askProjection(); return }
        try {
            batteryLauncher.launch(
                Intent(Settings.ACTION_REQUEST_IGNORE_BATTERY_OPTIMIZATIONS, Uri.parse("package:$packageName"))
            )
        } catch (e: Throwable) {
            AppLog.log("Battery dialog unavailable: ${e.message}")
            askProjection()
        }
    }

    private fun askProjection() {
        val mpm = getSystemService(MediaProjectionManager::class.java)
        projectionLauncher.launch(mpm.createScreenCaptureIntent())
    }

    private fun stopBroadcast() {
        startService(Intent(this, ScreenBroadcastService::class.java).setAction(ScreenBroadcastService.ACTION_STOP))
    }
}

@Composable
fun PublicTab(onOpen: (StreamInfo) -> Unit) {
    val streams = remember { mutableStateListOf<StreamInfo>() }
    DisposableEffect(Unit) {
        val ref = FirebaseDatabase.getInstance().getReference("streams")
        val l = object : ValueEventListener {
            override fun onDataChange(s: DataSnapshot) {
                streams.clear()
                for (c in s.children) {
                    val id = c.key ?: continue
                    if (c.child("status").value != "live") continue
                    streams.add(
                        StreamInfo(
                            id,
                            c.child("title").value as? String ?: "Untitled",
                            (c.child("viewerCount").value as? Number)?.toInt() ?: 0,
                            (c.child("startedAt").value as? Number)?.toLong() ?: 0L
                        )
                    )
                }
            }
            override fun onCancelled(e: DatabaseError) { AppLog.log("streams listener: ${e.message}") }
        }
        ref.addValueEventListener(l)
        onDispose { ref.removeEventListener(l) }
    }
    if (streams.isEmpty()) {
        Box(Modifier.fillMaxSize(), contentAlignment = Alignment.Center) { Text("No live broadcasts right now") }
    } else {
        LazyColumn(Modifier.fillMaxSize()) {
            items(streams) { s ->
                val mins = if (s.startedAt > 0) (System.currentTimeMillis() - s.startedAt) / 60000 else 0
                Card(Modifier.fillMaxWidth().padding(8.dp).clickable { onOpen(s) }) {

                        Row(verticalAlignment = Alignment.CenterVertically) {
                            Text("LIVE", color = Color.Red, fontWeight = FontWeight.Bold)
                            Spacer(Modifier.width(8.dp))
                            Text(s.title, fontWeight = FontWeight.SemiBold)
                        }
                        Text("${s.viewers} watching - live for $mins min", fontSize = 12.sp)
                    }
                }
            }
        }
    }
}

@Composable
fun ViewerScreen(info: StreamInfo, uid: String, onBack: () -> Unit) {
    val ctx = LocalContext.current
    val viewer = remember { Viewer(ctx.applicationContext, info.id, uid) }
    val state by viewer.state.collectAsState()
    val track by viewer.track.collectAsState()
    var renderer by remember { mutableStateOf<SurfaceViewRenderer?>(null) }

    BackHandler(onBack = onBack)
    DisposableEffect(Unit) {
        viewer.join()
        onDispose { viewer.leave() }
    }
    DisposableEffect(track, renderer) {
        val t = track
        val r = renderer
        if (t != null && r != null) t.addSink(r)
        onDispose {
            if (t != null && r != null) { try { t.removeSink(r) } catch (_: Throwable) {} }
        }
    }

    Column(Modifier.fillMaxSize()) {
        Row(Modifier.fillMaxWidth().padding(8.dp), verticalAlignment = Alignment.CenterVertically) {
            Button(onClick = onBack) { Text("Back") }
            Spacer(Modifier.width(12.dp))
            Column {
                Text(info.title, fontWeight = FontWeight.SemiBold)
                Text(state, fontSize = 12.sp)
            }
        }
        AndroidView(
            factory = { c ->
                SurfaceViewRenderer(c).also {
                    it.init(Rtc.eglBase.eglBaseContext, null)
                    it.setScalingType(RendererCommon.ScalingType.SCALE_ASPECT_FIT)
                    it.setEnableHardwareScaler(true)
                    renderer = it
                }
            },
            onRelease = { it.release() },
            modifier = Modifier.fillMaxWidth().weight(1f)
        )
    }
}

package com.screencastlive

import android.app.Notification
import android.app.NotificationChannel
import android.app.NotificationManager
import android.app.PendingIntent
import android.app.Service
import android.content.Context
import android.content.Intent
import android.content.pm.ServiceInfo
import android.os.Build
import android.os.IBinder
import android.os.PowerManager
import com.google.firebase.auth.FirebaseAuth
import kotlinx.coroutines.CoroutineScope
import kotlinx.coroutines.Dispatchers
import kotlinx.coroutines.SupervisorJob
import kotlinx.coroutines.cancel
import kotlinx.coroutines.launch

/**
 * Foreground service (type mediaProjection). Order that MUST be kept:
 *   1. startForeground(..., FOREGROUND_SERVICE_TYPE_MEDIA_PROJECTION)
 *   2. only then create ScreenCapturerAndroid and start capturing (inside Broadcaster.start()).
 * START_NOT_STICKY: the projection permission Intent is single-use on Android 14,
 * so after process death the user has to press Start again.
 */
class ScreenBroadcastService : Service() {
    companion object {
        const val ACTION_START = "com.screencastlive.START"
        const val ACTION_STOP = "com.screencastlive.STOP"
        const val EXTRA_DATA = "projection_data"
        const val EXTRA_TITLE = "title"
        private const val CHANNEL = "broadcast"
        private const val NOTIF_ID = 1
    }

    private var broadcaster: Broadcaster? = null
    private var wake: PowerManager.WakeLock? = null
    private val scope = CoroutineScope(SupervisorJob() + Dispatchers.Main)
    private var shuttingDown = false

    override fun onBind(intent: Intent?): IBinder? = null

    override fun onStartCommand(intent: Intent?, flags: Int, startId: Int): Int {
        when (intent?.action) {
            ACTION_STOP -> { shutdown("Stopped"); return START_NOT_STICKY }
            ACTION_START -> {}
            else -> { stopSelf(); return START_NOT_STICKY }
        }
        if (broadcaster != null || intent == null) return START_NOT_STICKY

        createChannel()
        AppLog.log("Service: startForeground(mediaProjection)")
        val n = buildNotification("Connecting")
        if (Build.VERSION.SDK_INT >= 29) {
            startForeground(NOTIF_ID, n, ServiceInfo.FOREGROUND_SERVICE_TYPE_MEDIA_PROJECTION)
        } else {
            startForeground(NOTIF_ID, n)
        }

        val data: Intent? = if (Build.VERSION.SDK_INT >= 33)
            intent.getParcelableExtra(EXTRA_DATA, Intent::class.java)
        else @Suppress("DEPRECATION") intent.getParcelableExtra(EXTRA_DATA)
        val title = intent.getStringExtra(EXTRA_TITLE) ?: "My screen"
        val user = FirebaseAuth.getInstance().currentUser

        if (data == null || user == null) {
            AppLog.log("Missing projection data or not signed in")
            shutdown("Error: missing permission data or not signed in")
            return START_NOT_STICKY
        }

        try {
            val pm = getSystemService(Context.POWER_SERVICE) as PowerManager
            wake = pm.newWakeLock(PowerManager.PARTIAL_WAKE_LOCK, "ScreenCastLive:broadcast").apply { acquire() }

            BroadcastState.state.value = "Starting"
            BroadcastState.running.value = true
            val b = Broadcaster(applicationContext, user.uid, title, data) { shutdown("Stopped") }
            broadcaster = b
            b.start()

            scope.launch {
                BroadcastState.state.collect { s ->
                    getSystemService(NotificationManager::class.java).notify(NOTIF_ID, buildNotification(s))
                }
            }
        } catch (e: Throwable) {
            AppLog.log("START FAILED: ${e.javaClass.simpleName}: ${e.message}")
            shutdown("Error: ${e.javaClass.simpleName}: ${e.message}")
        }
        return START_NOT_STICKY
    }

    private fun shutdown(finalState: String) {
        if (shuttingDown) return
        shuttingDown = true
        try { broadcaster?.stop() } catch (e: Throwable) { AppLog.log("stop error: ${e.message}") }
        broadcaster = null
        try { wake?.let { if (it.isHeld) it.release() } } catch (_: Throwable) {}
        BroadcastState.running.value = false
        BroadcastState.state.value = finalState
        scope.cancel()
        stopForeground(STOP_FOREGROUND_REMOVE)
        stopSelf()
    }

    override fun onDestroy() {
        if (!shuttingDown) shutdown("Stopped")
        super.onDestroy()
    }

    private fun createChannel() {
        val nm = getSystemService(NotificationManager::class.java)
        nm.createNotificationChannel(NotificationChannel(CHANNEL, "Broadcast", NotificationManager.IMPORTANCE_LOW))
    }

    private fun buildNotification(state: String): Notification {
        val stop = PendingIntent.getService(
            this, 0,
            Intent(this, ScreenBroadcastService::class.java).setAction(ACTION_STOP),
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        )
        val open = PendingIntent.getActivity(
            this, 1, Intent(this, MainActivity::class.java),
            PendingIntent.FLAG_IMMUTABLE or PendingIntent.FLAG_UPDATE_CURRENT
        )
        return Notification.Builder(this, CHANNEL)
            .setSmallIcon(android.R.drawable.ic_menu_camera)
            .setContentTitle("\uD83D\uDD34 LIVE - Broadcasting your screen")
            .setContentText(state)
            .setOngoing(true)
            .setContentIntent(open)
            .addAction(android.R.drawable.ic_media_pause, "Stop", stop)
            .build()
    }
}

package com.screencastlive

import android.content.Context
import com.google.firebase.database.ChildEventListener
import com.google.firebase.database.DataSnapshot
import com.google.firebase.database.DatabaseError
import com.google.firebase.database.FirebaseDatabase
import com.google.firebase.database.ValueEventListener
import kotlinx.coroutines.flow.MutableStateFlow
import org.webrtc.IceCandidate
import org.webrtc.MediaConstraints
import org.webrtc.PeerConnection
import org.webrtc.RtpTransceiver
import org.webrtc.SessionDescription
import org.webrtc.VideoTrack

/** Watches one broadcast. Viewer id == own uid. */
class Viewer(private val ctx: Context, private val streamId: String, private val uid: String) {
    val state = MutableStateFlow("Connecting...")
    val track = MutableStateFlow<VideoTrack?>(null)

    private val db = FirebaseDatabase.getInstance()
    private val ref = db.getReference("signaling/$streamId/viewers/$uid")
    private val statusRef = db.getReference("streams/$streamId/status")
    private var pc: PeerConnection? = null
    private var lastN = 0
    private var remoteSet = false
    private val pending = ArrayList<IceCandidate>()
    private var offerL: ValueEventListener? = null
    private var candL: ChildEventListener? = null
    private var statusL: ValueEventListener? = null
    private var errL: ValueEventListener? = null

    fun join() {
        Rtc.init(ctx)
        AppLog.log("Viewer: joining $streamId")
        ref.removeValue()
        ref.onDisconnect().removeValue()

        val conn = Rtc.factory.createPeerConnection(Rtc.rtcConfig(), object : PcObserver() {
            override fun onIceCandidate(c: IceCandidate?) {
                c ?: return
                ref.child("vc").push().setValue(
                    mapOf("mid" to c.sdpMid, "idx" to c.sdpMLineIndex, "sdp" to c.sdp)
                )
            }
            override fun onIceConnectionChange(s: PeerConnection.IceConnectionState?) {
                AppLog.log("Viewer ICE $s")
                when (s) {
                    PeerConnection.IceConnectionState.CONNECTED,
                    PeerConnection.IceConnectionState.COMPLETED -> state.value = "Live"
                    PeerConnection.IceConnectionState.DISCONNECTED -> state.value = "Connection interrupted, waiting..."
                    PeerConnection.IceConnectionState.FAILED -> state.value = "Connection failed, waiting for broadcaster to retry..."
                    else -> {}
                }
            }
            override fun onTrack(t: RtpTransceiver?) {
                val tr = t?.receiver?.track()
                if (tr is VideoTrack) {
                    AppLog.log("Viewer: video track received")
                    track.value = tr
                }
            }
        })
        if (conn == null) { state.value = "Error: could not create PeerConnection"; return }
        pc = conn

        offerL = object : ValueEventListener {
            override fun onDataChange(s: DataSnapshot) {
                val sdp = s.child("sdp").getValue(String::class.java) ?: return
                val n = (s.child("n").value as? Number)?.toInt() ?: return
                if (n <= lastN) return
                lastN = n
                val p = pc ?: return
                AppLog.log("Viewer: offer #$n received")
                p.setRemoteDescription(object : SdpAdapter("setRemote(offer)") {
                    override fun onSetSuccess() {
                        synchronized(pending) {
                            remoteSet = true
                            pending.forEach { p.addIceCandidate(it) }
                            pending.clear()
                        }
                        p.createAnswer(object : SdpAdapter("createAnswer") {
                            override fun onCreateSuccess(d: SessionDescription?) {
                                d ?: return
                                p.setLocalDescription(object : SdpAdapter("setLocal(answer)") {
                                    override fun onSetSuccess() {
                                        ref.child("answer").setValue(mapOf("sdp" to d.description, "n" to n))
                                        AppLog.log("Viewer: answer #$n sent")
                                    }
                                }, d)
                            }
                        }, MediaConstraints())
                    }
                }, SessionDescription(SessionDescription.Type.OFFER, sdp))
            }
            override fun onCancelled(e: DatabaseError) { AppLog.log("Viewer offer listener: ${e.message}") }
        }
        ref.child("offer").addValueEventListener(offerL!!)

        candL = object : ChildEventListener {
            override fun onChildAdded(s: DataSnapshot, prev: String?) {
                val sdp = s.child("sdp").getValue(String::class.java) ?: return
                val mid = s.child("mid").getValue(String::class.java)
                val idx = (s.child("idx").value as? Number)?.toInt() ?: 0
                val cand = IceCandidate(mid, idx, sdp)
                synchronized(pending) {
                    if (remoteSet) pc?.addIceCandidate(cand) else pending.add(cand)
                }
            }
            override fun onChildChanged(s: DataSnapshot, prev: String?) {}
            override fun onChildRemoved(s: DataSnapshot) {}
            override fun onChildMoved(s: DataSnapshot, prev: String?) {}
            override fun onCancelled(e: DatabaseError) {}
        }
        ref.child("bc").addChildEventListener(candL!!)

        errL = object : ValueEventListener {
            override fun onDataChange(s: DataSnapshot) {
                if (s.value == "full") state.value = "Viewer limit reached for this stream"
            }
            override fun onCancelled(e: DatabaseError) {}
        }
        ref.child("error").addValueEventListener(errL!!)

        statusL = object : ValueEventListener {
            override fun onDataChange(s: DataSnapshot) {
                if (s.value == "offline") state.value = "Broadcaster is offline / reconnecting..."
            }
            override fun onCancelled(e: DatabaseError) {}
        }
        statusRef.addValueEventListener(statusL!!)

        ref.child("join").setValue(true)
    }

    fun leave() {
        AppLog.log("Viewer: leaving")
        track.value = null
        offerL?.let { ref.child("offer").removeEventListener(it) }
        candL?.let { ref.child("bc").removeEventListener(it) }
        errL?.let { ref.child("error").removeEventListener(it) }
        statusL?.let { statusRef.removeEventListener(it) }
        try { pc?.close(); pc?.dispose() } catch (_: Throwable) {}
        pc = null
        ref.removeValue()
        ref.onDisconnect().cancel()
    }
}
